// 自定义RSI指标
indicator("自定义RSI", overlay=false)

// 参数配置
length = input(14, "周期")
overbought = input(70, "超买线")
oversold = input(30, "超卖线")

// 计算RSI
change = close - close[1]
gain = change >= 0 ? change : 0
loss = change < 0 ? -change : 0

avgGain = sma(gain, length)
avgLoss = sma(loss, length)

rs = avgGain / avgLoss
rsi = 100 - (100 / (1 + rs))

// 绘制
plot(rsi, "RSI", color.blue, 2)
hline(overbought, "超买", color.red, 1)
hline(oversold, "超卖", color.green, 1)
hline(50, "中线", color.gray, 1)

// 信号
buySignal = crossover(rsi, oversold)
sellSignal = crossunder(rsi, overbought)

plotshape(buySignal, "买入", shape.triangleup,
          location.bottom, color.green, size=size.small)
plotshape(sellSignal, "卖出", shape.triangledown,
          location.top, color.red, size=size.small)
