import { execSync } from 'child_process';

export interface AgentOSResult {
  success: boolean;
  data?: any;
  error?: string;
}

/**
 * Execute agent-os CLI command
 */
export async function execAgentOS(args: string[]): Promise<AgentOSResult> {
  try {
    const agentOSPath = process.env.AGENT_OS_PATH || 
      '/Users/yunpeng/pi-investment/agent-os/agent-os';
    
    const cmd = `${agentOSPath} ${args.join(' ')}`;
    
    console.log(`[agent-os-cli] Executing: ${cmd}`);
    
    const stdout = execSync(cmd, {
      encoding: 'utf-8',
      stdio: ['pipe', 'pipe', 'pipe'],
      timeout: 30000, // 30s timeout
    });
    
    // Try to parse JSON output
    try {
      const data = JSON.parse(stdout.trim());
      return { success: true, data };
    } catch {
      // Not JSON, return as string
      return { success: true, data: stdout.trim() };
    }
  } catch (error: any) {
    console.error(`[agent-os-cli] Error:`, error.message);
    return {
      success: false,
      error: error.message,
    };
  }
}

/**
 * Check if agent-os is available
 */
export async function checkAgentOS(): Promise<boolean> {
  try {
    const result = await execAgentOS(['version']);
    return result.success;
  } catch {
    return false;
  }
}

/**
 * Memory operations
 */
export const AgentOSMemory = {
  async write(params: {
    content: string;
    category: string;
    importance?: number;
    agentId?: string;
  }): Promise<any> {
    const { content, category, importance = 5, agentId = 'fin-agent' } = params;
    
    return execAgentOS([
      'memory', 'write',
      '--content', content,
      '--category', category,
      '--importance', importance.toString(),
      '--agent-id', agentId,
      '--json',
    ]);
  },

  async search(params: {
    query: string;
    agentId?: string;
    topK?: number;
  }): Promise<any> {
    const { query, agentId = 'fin-agent', topK = 10 } = params;
    
    return execAgentOS([
      'memory', 'search',
      '--query', query,
      '--agent-id', agentId,
      '--top-k', topK.toString(),
      '--json',
    ]);
  },

  async recallAudit(params: {
    agentId?: string;
    dateFrom?: string;
    dateTo?: string;
  }): Promise<any> {
    const { agentId = 'fin-agent', dateFrom, dateTo } = params;
    
    const args = [
      'memory', 'recall-audit',
      '--agent-id', agentId,
      '--json',
    ];
    
    if (dateFrom) args.push('--date-from', dateFrom);
    if (dateTo) args.push('--date-to', dateTo);
    
    return execAgentOS(args);
  },
};

/**
 * Decision operations
 */
export const AgentOSDecision = {
  async record(params: {
    action: string;
    reason: string;
    confidence: number;
    agentId?: string;
    metadata?: Record<string, any>;
  }): Promise<any> {
    const { action, reason, confidence, agentId = 'fin-agent', metadata = {} } = params;
    
    return execAgentOS([
      'decision', 'record',
      '--action', action,
      '--reason', reason,
      '--confidence', confidence.toString(),
      '--agent-id', agentId,
      '--metadata', JSON.stringify(metadata),
      '--json',
    ]);
  },
};

 * Notification operations

export const AgentOSNotification = {
  async send(params: {
    channel: string;
    title: string;

```typescript
import type {
  MemoryProvider,
  MemorySearchResult,
  MemorySearchResponse,
  MemoryWriteParams,
  ExperienceWriteParams,
} from './port.js';
import { AgentOSMemory } from '../../infrastructure/agent-os/cli.js';

export class AgentOSMemoryProvider implements MemoryProvider {
  readonly name = 'agent-os-memory';

  private sessionId: string = '';
  private sessionKind: string = 'user';
  private agentId: string = 'fin-agent';
  private initialized = false;

  constructor() {}

  isAvailable(): boolean {
    return true; // Agent OS 总是可用
  }

  async initialize(
    sessionId: string,
    context: {
      sessionKind?: string;
      agentId?: string;
    }
  ): Promise<void> {
    this.sessionId = sessionId;
    this.sessionKind = context.sessionKind || 'user';
    this.agentId = context.agentId || 'fin-agent';
    this.initialized = true;
  }

  async write(params: MemoryWriteParams): Promise<void> {
    if (!this.initialized) {
      throw new Error('AgentOSMemoryProvider not initialized');
    }

    const result = await AgentOSMemory.write({
      content: params.content,
      category: params.category,
      importance: params.importance || 5,
      agentId: this.agentId,
    });

    if (!result.success) {
      throw new Error(`Memory write failed: ${result.error}`);
    }
  }

  async writeExperience(params: ExperienceWriteParams): Promise<void> {
    // Experience 本质上也是记忆，只是 category 不同
    await this.write({
      content: params.observation,
      category: 'experience',
      importance: params.impact ? 8 : 5,
    });
  }

  async search(query: string, topK: number = 10): Promise<MemorySearchResponse> {
    if (!this.initialized) {
      throw new Error('AgentOSMemoryProvider not initialized');
    }

    const result = await AgentOSMemory.search({
      query,
      agentId: this.agentId,
      topK,
    });

    if (!result.success) {
      throw new Error(`Memory search failed: ${result.error}`);
    }

    // 转换 Agent OS 格式到 agent-ts 格式
    const memories: MemorySearchResult[] = (result.data || []).map((m: any) => ({
      id: m.id,
      content: m.content,
      category: m.category,
      importance: m.importance,
      score: m.score,
      createdAt: m.created_at,
      metadata: m.metadata || {},
    }));

    return {
      memories,
      total: memories.length,
    };
  }

  async recallAudit(dateFrom?: string, dateTo?: string): Promise<any> {
    if (!this.initialized) {
      throw new Error('AgentOSMemoryProvider not initialized');
    }

    const result = await AgentOSMemory.recallAudit({
      agentId: this.agentId,
      dateFrom,
      dateTo,
    });

    if (!result.success) {
      throw new Error(`Recall audit failed: ${result.error}`);
    }

    return result.data;
  }

  async validateRecall(memoryIds: number[]): Promise<void> {
    // Agent OS 自动处理 recall validation
    // 不需要额外操作
  }

  async syncTurn(content: string): Promise<void> {
    // Agent OS 不需要 syncTurn
    // 每次 write/search 都是独立的
  }
}
```

### Phase C: 更新 Memory Provider Manager（15 分钟）

#### Step 3: 修改 provider-manager.ts

**位置**: `src/services/memory/provider-manager.ts`

```typescript
import { V2MemoryProvider } from './v2-client.js';
import { AgentOSMemoryProvider } from './agent-os-provider.js';
import { FileFallbackProvider } from './file-fallback.js';
import type { MemoryProvider } from './port.js';

export function getMemoryProvider(): MemoryProvider {
  // 优先使用 Agent OS
  const useAgentOS = process.env.USE_AGENT_OS !== 'false';
  
  if (useAgentOS) {
    console.log('[MemoryProvider] Using Agent OS Memory');
    return new AgentOSMemoryProvider();
  }

  // 降级到 v2（兼容模式）
  const v2Provider = new V2MemoryProvider();
  if (v2Provider.isAvailable()) {
    console.log('[MemoryProvider] Using V2 Memory (fallback)');
    return v2Provider;
  }

  // 最后降级到文件系统
  console.warn('[MemoryProvider] Falling back to file-based memory');
  return new FileFallbackProvider();
}
```

import { AgentOSNotification } from '../agent-os/cli.js';

export interface NotificationParams {
  channel: 'trading' | 'alerts' | 'reports';
  title: string;
  content: string;
  priority?: 'low' | 'normal' | 'high';
}

export class AgentOSNotificationService {
  async send(params: NotificationParams): Promise<void> {
    const result = await AgentOSNotification.send({
      channel: params.channel,
      title: params.title,
      content: params.content,
      priority: params.priority || 'normal',
    });

    if (!result.success) {
      throw new Error(`Notification send failed: ${result.error}`);
    }
  }

  async sendTrading(title: string, content: string): Promise<void> {
    await this.send({
      channel: 'trading',
      title,
      content,
    });
  }

  async sendAlert(title: string, content: string): Promise<void> {
    await this.send({
      channel: 'alerts',
      title,
      content,
      priority: 'high',
    });
  }

  async sendReport(title: string, content: string): Promise<void> {
    await this.send({
      channel: 'reports',
      title,
      content,
    });
  }
}

// Export singleton
export const agentOSNotification = new AgentOSNotificationService();

## 🚨 注意事项

### 1. 兼容性处理

保留 v2 provider 作为降级方案：

```typescript
// 如果 Agent OS 不可用，自动降级
if (!agentOSProvider.isAvailable()) {
  return new V2MemoryProvider();
}
```

### 2. 错误处理

所有 CLI 调用都需要 try-catch：

```typescript
try {
  await AgentOSMemory.write(...);
} catch (error) {
  console.error('[AgentOS] Memory write failed:', error);
  // 降级到文件系统？或抛出错误？
}
```

### 3. 性能考虑

CLI 调用有额外开销（进程启动），可以考虑：
- 缓存常用结果
- 批量操作
- 未来改为 HTTP API（如果性能成为瓶颈）
