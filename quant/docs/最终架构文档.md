### 使用
```python
# 数据获取
from quantsys.data.fetchers import klines
data = klines.fetch_daily_klines('000001', days=730)

# 因子计算
from quantsys.factors import MA, RSI
ma5 = MA(5).calculate(data)
rsi14 = RSI(14).calculate(data)

# 策略回测
from quantsys.backtest import BacktestEngine
from quantsys.strategies.classic import RSIReversalStrategy

engine = BacktestEngine()
strategy = RSIReversalStrategy()
result = engine.run(strategy, data, '2024-01-01', '2024-12-31')

# 风控管理
from quantsys.risk import PreTradeRiskCheck
risk_check = PreTradeRiskCheck()
is_valid, error = risk_check.check(order, portfolio)
```
